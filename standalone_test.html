Usage Example

<!-- 1. Include the module -->
<script src="dist/contour-core.umd.js"></script>

<!-- 2. Use it -->
<script>
// Prepare your grid data
var grid = {
    z: [[0,1,2],[1,2,3],[2,3,4]],  // 2D array of values
    x: [0,1,2],                      // X coordinates (optional)
    y: [0,1,2]                       // Y coordinates (optional)
};

// Compute contours
var result = ContourCore.computeContours(grid, {
    autocontour: true,
    ncontours: 10,
    smoothing: 0.5
});

// Result contains:
console.log(result.levels);  // Array of contour level values
console.log(result.paths);   // Array of path objects
</script>
