Overlay 交互式绘制 (ES Module)

使用 ES6 import 语法加载 contour-core Overlay 系统

// ES6 模块导入
import contourCore from '../dist/contour-core.esm.mjs';

// 解构获取渲染器和事件枚举
const { computeContours, renderers, DrawEvents, DrawStatus } = contourCore;

// 获取 Overlay 系统
const overlay = renderer.getOverlay();
const status = overlay.getDrawStatus(); // 'idle' | 'drawing' | 'completed'
