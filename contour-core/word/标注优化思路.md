4. **闭合路径处理不完整** - 环形等值线标注位置不准确

## 优化方案

### 优化点 1：完善线段相交检测

**问题**：当前 `segmentDistance` 只计算端点到线段的距离，未检测线段相交

**解决方案**：参考 `等值线中标注实现的原理.md` 第 265-283 行

```javascript
// 在 labels/cost.js 中添加
function segmentsIntersect(x1, y1, x2, y2, x3, y3, x4, y4) {
    var a = x2 - x1, b = x3 - x1, c = x4 - x3;
    var d = y2 - y1, e = y3 - y1, f = y4 - y3;
    var det = a * f - c * d; // 行列式

    if (det === 0) return null; // 平行线

    var t = (b * f - c * e) / det;
    var u = (b * d - a * e) / det;

    // t和u都在[0,1]范围内时线段相交
    if (u < 0 || u > 1 || t < 0 || t > 1) return null;

    return { x: x1 + a * t, y: y1 + d * t };
}

// 修改 segmentDistance
function segmentDistance(x1, y1, x2, y2, x3, y3, x4, y4) {
    // 1. 如果线段相交，距离为0
    if (segmentsIntersect(x1, y1, x2, y2, x3, y3, x4, y4)) return 0;

    // 2. 计算线段向量和长度平方
    var x12 = x2 - x1, y12 = y2 - y1;
    var x34 = x4 - x3, y34 = y4 - y3;
    var ll12 = x12 * x12 + y12 * y12;
    var ll34 = x34 * x34 + y34 * y34;

    // 3. 计算四个端点到另一线段的最小距离
    var dist2 = Math.min(
        perpDistance2(x12, y12, ll12, x3 - x1, y3 - y1),
        perpDistance2(x12, y12, ll12, x4 - x1, y4 - y1),
        perpDistance2(x34, y34, ll34, x1 - x3, y1 - y3),
        perpDistance2(x34, y34, ll34, x2 - x3, y2 - y3)
    );

    return Math.sqrt(dist2);
}
```

**预期效果**：更精确的防碰撞检测，避免交叉标注重叠

### 优化点 2：闭合路径环绕处理

**问题**：`getTextLocation` 未考虑闭合路径的环绕特性

**解决方案**：参考原理文档第 179-205 行，添加 mod 操作

```javascript
// 在 labels/position.js 中修改
function mod(n, m) {
    return ((n % m) + m) % m;
}

function getTextLocation(path, totalPathLen, positionOnPath, textWidth, isClosed) {
    // 对于闭合路径，使用 mod 操作处理环绕
    var halfWidth = textWidth / 2;
    var p0Pos, p1Pos;

    if (isClosed) {
        p0Pos = mod(positionOnPath - halfWidth, totalPathLen);
        p1Pos = mod(positionOnPath + halfWidth, totalPathLen);
    } else {
        p0Pos = Math.max(0, positionOnPath - halfWidth);
        p1Pos = Math.min(totalPathLen, positionOnPath + halfWidth);
    }

    var p0 = getPointAtLength(path, p0Pos);
    var p1 = getPointAtLength(path, p1Pos);
    var pCenter = getPointAtLength(path, positionOnPath);

    // 计算角度和位置
    var theta = Math.atan2(p1.y - p0.y, p1.x - p0.x);
    var x = (pCenter.x * 4 + p0.x + p1.x) / 6;
    var y = (pCenter.y * 4 + p0.y + p1.y) / 6;

    return { x: x, y: y, theta: theta };
}
```

**预期效果**：环形等值线标注位置更准确，不会因路径环绕而错误计算

### 优化点 4：位置缓存机制

**问题**：重复计算同一路径位置

**解决方案**：参考原理文档第 180-188 行

```javascript
// 在 labels/position.js 中添加
var workingPath = null;
var workingTextWidth = 0;
var locationCache = {};

function getTextLocation(path, totalPathLen, positionOnPath, textWidth, isClosed) {
    // 使用短期缓存优化性能
    if (path !== workingPath || textWidth !== workingTextWidth) {
        locationCache = {};
        workingPath = path;
        workingTextWidth = textWidth;
    }

    // 使用位置作为缓存键（保留适当精度）
    var cacheKey = Math.round(positionOnPath * 100) / 100;
    if (locationCache[cacheKey] !== undefined) {
        return locationCache[cacheKey];
    }

    // ... 计算位置

    locationCache[cacheKey] = result;
    return result;
}
```

**预期效果**：减少重复计算，提升性能

### 优化点 6：完善垂直距离计算

**问题**：当前 `pointToSegmentDistance` 实现较简化

**解决方案**：参考原理文档第 244-263 行

```javascript
// 在 labels/cost.js 中替换
function perpDistance2(xab, yab, llab, xac, yac) {
    var fcAB = (xac * xab + yac * yab); // 点积
    if (fcAB < 0) {
        // 点c在线段ab的a端外侧
        return xac * xac + yac * yac;
    } else if (fcAB > llab) {
        // 点c在线段ab的b端外侧
        var xbc = xac - xab;
        var ybc = yac - yab;
        return xbc * xbc + ybc * ybc;
    } else {
        // 点c在线段ab上的投影距离（垂直距离）
        var crossProduct = xac * yab - yac * xab;
        return crossProduct * crossProduct / llab;
    }
}
```

## 总结

本次优化的核心是：

1. **补全缺失的几何算法** - 线段相交、闭合路径处理
2. **实现完整的密度控制** - 多标注、自适应数量
3. **修复坐标系统不一致** - 统一使用网格坐标计算
4. **性能优化** - 缓存、搜索范围调整

通过以上优化，contour-core 的标注系统将达到与 Plotly 原版相当的效果，解决标注压盖问题。
