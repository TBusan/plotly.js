## 概述

等值线标注是指在等值线图上沿等值线路径显示数值标签的功能。Plotly.js 的标注实现具有以下特点：

- **沿路径对齐**：标注沿等值线的切线方向旋转，与线条保持一致
- **智能防碰撞**：通过成本函数避免标注重叠
- **自适应密度**：根据等值线长度和级别数量自动调整标注数量
- **边缘规避**：优先选择远离图表边缘的位置放置标注

## 标注生成流程

### 主流程概述

标注生成的入口是 `makeLinesAndLabels` 函数（`src/traces/contour/plot.js:202`）：

```
┌─────────────────────────────────────────────────────────────┐
│                    makeLinesAndLabels                        │
├─────────────────────────────────────────────────────────────┤
│  1. 创建等值线路径 (createLines)                              │
│  2. 计算可见区域边界 (bounds)                                 │
│  3. 初始化标注格式化器 (labelFormatter)                       │
│  4. 遍历每条等值线路径                                         │
│     ├─ 计算文本尺寸 (calcTextOpts)                           │
│     ├─ 获取可见路径段 (getVisibleSegment)                    │
│     ├─ 计算最大标注数 (maxLabels)                            │
│     └─ 循环放置标注 (findBestTextLocation + addLabelData)    │
│  5. 渲染所有标注 (drawLabels)                                │
└─────────────────────────────────────────────────────────────┘
```

### 详细代码流程

```javascript
// src/traces/contour/plot.js:202-320
function makeLinesAndLabels(plotgroup, pathinfo, gd, cd0, contours) {
    // 1. 判断是否显示标注
    var showLabels = contours.showlabels;
    if (!showLabels) return;

    // 2. 创建等值线路径（即使不显示线条也要创建，用于定位标注）
    var linegroup = exports.createLines(lineContainer, true, pathinfo, isStatic);

    // 3. 计算绘图边界
    var bounds = {
        left: x0, right: x1,
        top: y0, bottom: y1,
        middle: (top + bottom) / 2,
        center: (left + right) / 2
    };

    // 4. 初始化数据
    var labelData = [];
    var labelClipPathData = [];
    var contourFormat = exports.labelFormatter(gd, cd0);
    var dummyText = Drawing.tester.append('text'); // 用于测量文本尺寸

    // 5. 遍历每条等值线
    linegroup.each(function(d) {
        var textOpts = exports.calcTextOpts(d.level, contourFormat, dummyText, gd);

        d3.select(this).selectAll('path').each(function() {
            var path = this;
            // 5.1 获取可见路径段
            var pathBounds = Lib.getVisibleSegment(path, bounds, textOpts.height / 2);
            if (!pathBounds) return;

            // 5.2 路径太短则不放置标注
            if (pathBounds.len < (textOpts.width + textOpts.height) * constants.LABELMIN) {
                return;
            }

            // 5.3 计算最大标注数
            var maxLabels = Math.min(
                Math.ceil(pathBounds.len / normLength),
                constants.LABELMAX
            );

            // 5.4 循环放置标注
            for (var i = 0; i < maxLabels; i++) {
                var loc = exports.findBestTextLocation(
                    path, pathBounds, textOpts, labelData, bounds
                );
                if (!loc) break; // 找不到合适位置则停止
                exports.addLabelData(loc, textOpts, labelData, labelClipPathData);
            }
        });
    });

    // 6. 渲染所有标注
    exports.drawLabels(labelGroup, labelData, gd, lineClip, labelClipPathData);
}
```

## 位置计算算法

### getTextLocation 函数

这是标注位置计算的核心函数（`src/lib/geometry2d.js:83-108`）：

```javascript
exports.getTextLocation = function getTextLocation(path, totalPathLen, positionOnPath, textWidth) {
    // 使用短期缓存优化性能
    if (path !== workingPath || textWidth !== workingTextWidth) {
        locationCache = {};
        workingPath = path;
        workingTextWidth = textWidth;
    }
    if (locationCache[positionOnPath]) {
        return locationCache[positionOnPath];
    }

    // 1. 获取文本宽度两端点，用于计算角度
    var p0 = path.getPointAtLength(mod(positionOnPath - textWidth / 2, totalPathLen));
    var p1 = path.getPointAtLength(mod(positionOnPath + textWidth / 2, totalPathLen));

    // 2. 计算角度：沿等值线的切线方向
    var theta = Math.atan((p1.y - p0.y) / (p1.x - p0.x));

    // 3. 计算文本中心位置（加权平均）
    // 使用 2/3 的中心位置 + 1/3 的端点中点
    // 假设路径段大致是二次曲线
    var pCenter = path.getPointAtLength(mod(positionOnPath, totalPathLen));
    var x = (pCenter.x * 4 + p0.x + p1.x) / 6;
    var y = (pCenter.y * 4 + p0.y + p1.y) / 6;

    return { x: x, y: y, theta: theta };
};
```

**关键点**：
- 使用 SVG `path.getPointAtLength()` API 获取路径上的精确坐标
- 角度由文本宽度两端点确定，确保文本与路径对齐
- 位置采用加权平均（4:1:1），考虑了曲线弯曲

## 防碰撞检测算法

### 1. 线段距离计算 (segmentDistance)

防碰撞的基础是计算两个线段之间的最小距离（`src/lib/geometry2d.js:32-52`）：

```javascript
exports.segmentDistance = function segmentDistance(x1, y1, x2, y2, x3, y3, x4, y4) {
    // 1. 如果线段相交，距离为0
    if (segmentsIntersect(x1, y1, x2, y2, x3, y3, x4, y4)) return 0;

    // 2. 计算线段向量和长度平方
    var x12 = x2 - x1, y12 = y2 - y1;
    var x34 = x4 - x3, y34 = y4 - y3;
    var ll12 = x12 * x12 + y12 * y12;
    var ll34 = x34 * x34 + y34 * y34;

    // 3. 计算四个端点到另一线段的最小距离
    var dist2 = Math.min(
        perpDistance2(x12, y12, ll12, x3 - x1, y3 - y1),  // 点3到线段12
        perpDistance2(x12, y12, ll12, x4 - x1, y4 - y1),  // 点4到线段12
        perpDistance2(x34, y34, ll34, x1 - x3, y1 - y3),  // 点1到线段34
        perpDistance2(x34, y34, ll34, x2 - x3, y2 - y3)   // 点2到线段34
    );

    return Math.sqrt(dist2);
};
```

**垂直距离计算**（点到线段的最短距离）：

```javascript
function perpDistance2(xab, yab, llab, xac, yac) {
    var fcAB = (xac * xab + yac * yab); // 点积
    if (fcAB < 0) {
        // 点c在线段ab的a端外侧
        return xac * xac + yac * yac;
    } else if (fcAB > llab) {
        // 点c在线段ab的b端外侧
        var xbc = xac - xab;
        var ybc = yac - yab;
        return xbc * xbc + ybc * ybc;
    } else {
        // 点c在线段ab上的投影距离（垂直距离）
        var crossProduct = xac * yab - yac * xab;
        return crossProduct * crossProduct / llab;
    }
}
```

### 2. 线段相交检测 (segmentsIntersect)

```javascript
function segmentsIntersect(x1, y1, x2, y2, x3, y3, x4, y4) {
    var a = x2 - x1, b = x3 - x1, c = x4 - x3;
    var d = y2 - y1, e = y3 - y1, f = y4 - y3;
    var det = a * f - c * d; // 行列式

    if (det === 0) return null; // 平行线

    var t = (b * f - c * e) / det;
    var u = (b * d - a * e) / det;

    // t和u都在[0,1]范围内时线段相交
    if (u < 0 || u > 1 || t < 0 || t > 1) return null;

    return { x: x1 + a * t, y: y1 + d * t };
}
```

## 成本优化函数

标注放置的核心是成本函数 `locationCost`（`src/traces/contour/plot.js:490-538`）：

```javascript
function locationCost(loc, textOpts, labelData, bounds) {
    var halfWidth = textOpts.width / 2;
    var halfHeight = textOpts.height / 2;
    var x = loc.x, y = loc.y, theta = loc.theta;
    var dx = Math.cos(theta) * halfWidth;
    var dy = Math.sin(theta) * halfWidth;

    // =====================
    // 成本1：边缘惩罚 (EDGECOST)
    // =====================
    // 计算标注到各边缘的归一化距离
    var normX = ((x > bounds.center) ?
        (bounds.right - x) : (x - bounds.left)) /
        (dx + Math.abs(Math.sin(theta) * halfHeight));
    var normY = ((y > bounds.middle) ?
        (bounds.bottom - y) : (y - bounds.top)) /
        (Math.abs(dy) + Math.cos(theta) * halfHeight);

    // 如果超出边界，返回无穷大
    if (normX < 1 || normY < 1) return Infinity;

    // 边缘成本 = EDGECOST * (1/(normX-1) + 1/(normY-1))
    var cost = costConstants.EDGECOST * (1 / (normX - 1) + 1 / (normY - 1));

    // =====================
    // 成本2：角度惩罚 (ANGLECOST)
    // =====================
    // 鼓励水平放置（theta=0时成本为0）
    cost += costConstants.ANGLECOST * theta * theta;

    // =====================
    // 成本3：邻居惩罚 (NEIGHBORCOST)
    // =====================
    var x1 = x - dx, y1 = y - dy;  // 标注线段起点
    var x2 = x + dx, y2 = y + dy;  // 标注线段终点

    for (var i = 0; i < labelData.length; i++) {
        var labeli = labelData[i];
        var dxd = Math.cos(labeli.theta) * labeli.width / 2;
        var dyd = Math.sin(labeli.theta) * labeli.width / 2;

        // 计算两个标注线段之间的距离（归一化）
        var dist = Lib.segmentDistance(
            x1, y1, x2, y2,
            labeli.x - dxd, labeli.y - dyd,
            labeli.x + dxd, labeli.y + dyd
        ) * 2 / (textOpts.height + labeli.height);

        var sameLevel = labeli.level === textOpts.level;
        var distOffset = sameLevel ? costConstants.SAMELEVELDISTANCE : 1;

        // 距离太近直接返回无穷大
        if (dist <= distOffset) return Infinity;

        // 邻居成本 = NEIGHBORCOST * 系数 / (dist - offset)
        var distFactor = costConstants.NEIGHBORCOST *
            (sameLevel ? costConstants.SAMELEVELFACTOR : 1);
        cost += distFactor / (dist - distOffset);
    }

    return cost;
}
```

### 成本函数可视化

```
成本 = 边缘成本 + 角度成本 + 邻居成本

        │
   高成本 │ (靠近边缘)
        │
        │    低成本区域
        │    (远离边缘 + 水平 + 无邻居)
        │
        └───────────────────
           靠近边缘  远离边缘
```

## 渲染与显示

### 1. 标注数据添加 (addLabelData)

```javascript
// src/traces/contour/plot.js:540-578
exports.addLabelData = function(loc, textOpts, labelData, labelClipPathData) {
    var fontSize = textOpts.fontSize;
    var w = textOpts.width + fontSize / 3;   // 添加边距
    var h = Math.max(0, textOpts.height - fontSize / 3);

    var x = loc.x, y = loc.y, theta = loc.theta;
    var sin = Math.sin(theta), cos = Math.cos(theta);

    // 旋转坐标变换
    var rotateXY = function(dx, dy) {
        return [
            x + dx * cos - dy * sin,
            y + dx * sin + dy * cos
        ];
    };

    // 计算旋转后的边界框四个顶点
    var bBoxPts = [
        rotateXY(-w / 2, -h / 2),  // 左上
        rotateXY(-w / 2, h / 2),   // 左下
        rotateXY(w / 2, h / 2),    // 右下
        rotateXY(w / 2, -h / 2)    // 右上
    ];

    labelData.push({
        text: textOpts.text,
        x: x, y: y,
        dy: textOpts.dy,
        theta: theta,
        level: textOpts.level,
        width: w, height: h
    });

    // 保存边界框用于裁剪
    labelClipPathData.push(bBoxPts);
};
```

### 2. 标注绘制 (drawLabels)

```javascript
// src/traces/contour/plot.js:580-615
exports.drawLabels = function(labelGroup, labelData, gd, lineClip, labelClipPathData) {
    var labels = labelGroup.selectAll('text')
        .data(labelData, function(d) {
            return d.text + ',' + d.x + ',' + d.y + ',' + d.theta;
        });

    labels.exit().remove();

    labels.enter().append('text')
        .attr({
            'data-notex': 1,
            'text-anchor': 'middle'
        })
        .each(function(d) {
            // 计算最终位置（考虑dy偏移）
            var x = d.x + Math.sin(d.theta) * d.dy;
            var y = d.y - Math.cos(d.theta) * d.dy;

            d3.select(this)
                .text(d.text)
                .attr({
                    x: x,
                    y: y,
                    transform: 'rotate(' +
                        (180 * d.theta / Math.PI) + ' ' +
                        x + ' ' + y + ')'
                })
                .call(svgTextUtils.convertToTspans, gd);
        });

    // 创建裁剪路径（隐藏标注下方的等值线）
    if (labelClipPathData) {
        var clipPath = '';
        for (var i = 0; i < labelClipPathData.length; i++) {
            clipPath += 'M' + labelClipPathData[i].join('L') + 'Z';
        }
        var lineClipPath = Lib.ensureSingle(lineClip, 'path', '');
        lineClipPath.attr('d', clipPath);
    }
};
```

## 算法总结

### 标注放置的迭代搜索

`findBestTextLocation` 函数使用迭代优化找到最佳位置：

```javascript
// src/traces/contour/plot.js:447-481
exports.findBestTextLocation = function(path, pathBounds, textOpts, labelData, plotBounds) {
    // 1. 初始搜索点分布
    var dp = pathBounds.len / costConstants.INITIALSEARCHPOINTS;
    var p0 = pathBounds.min + dp / 2;
    var pMax = pathBounds.max;

    var cost = Infinity;

    // 2. 多轮迭代优化
    for (var j = 0; j < costConstants.ITERATIONS; j++) {
        for (var p = p0; p < pMax; p += dp) {
            var newLocation = Lib.getTextLocation(path, pathBounds.total, p, textWidth);
            var newCost = locationCost(newLocation, textOpts, labelData, plotBounds);

            if (newCost < cost) {
                cost = newCost;
                loc = newLocation;
                pMin = p;
            }
        }

        if (cost > costConstants.MAXCOST * 2) break;

        // 3. 二分细化搜索
        if (j) dp /= 2;
        p0 = pMin - dp / 2;
        pMax = p0 + dp * 1.5;
    }

    if (cost <= costConstants.MAXCOST) return loc;
};
```

### 搜索过程示意

```
迭代1 (粗搜索):  ●   ●   ●   ●   ●   ●   ●   ●   ●   ●
                   ↑
                 找到最低成本点

迭代2 (细搜索):      ● ● ● ● ●
                       ↑
                     精细化最低点

迭代3-5: 继续细化...
```

## 关键设计思想

1. **解耦计算与渲染**：标注位置计算是纯数学运算，不依赖DOM
2. **成本驱动优化**：通过统一的成本函数综合多个约束条件
3. **渐进式搜索**：先粗后细的迭代策略保证效率
4. **几何精确性**：使用线段距离而非简单点距离
5. **视觉友好**：鼓励水平放置、远离边缘、保持合理间距
