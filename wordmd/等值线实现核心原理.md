# Plotly.js Contour Fill 模式实现核心原理

## 一、概述

Plotly.js 的 contour 实现使用改进的 **Marching Squares 算法**来从二维网格数据中提取等值线和等值面。在 `fill` 模式下，系统不仅绘制等值线，还会填充等值线之间的区域，形成等值面效果。

## 二、核心文件结构

```

├── plot.js              # 渲染主逻辑，处理 SVG/Canvas 绘制
```

### 3.2 改进的 Marching Squares 算法

#### 3.2.1 基本原理

Marching Squares 将网格分成 2×2 的单元格，每个单元格的 4 个角点与等值线层级比较，生成一个 4 位索引（0-15）。

```javascript
// make_crossings.js
function getMarchingIndex(val, corners) {
    // corners 格式：[[左上, 右上], [左下, 右下]]
    var mi = (corners[0][0] > val ? 0 : 1) +   // 左上
             (corners[0][1] > val ? 0 : 2) +   // 右上
             (corners[1][1] > val ? 0 : 4) +   // 右下
             (corners[1][0] > val ? 0 : 8);    // 左下

    // 处理鞍点（ambiguous cases）
    if(mi === 5 || mi === 10) {
        var avg = (corners[0][0] + corners[0][1] +
                   corners[1][0] + corners[1][1]) / 4;
        if(val > avg) return (mi === 5) ? 713 : 1114;  // 双峰
        return (mi === 5) ? 104 : 208;  // 双谷
    }
    return (mi === 15) ? 0 : mi;
}
```

**鞍点处理的关键**：
- 索引 5 和 10 是鞍点情况（两条可能的等值线）
- 通过比较层级值与四个角点的平均值来决定连接方式
- 使用复合索引（如 104, 208, 713, 1114）表示两种可能性的组合

#### 3.2.2 交点计算与插值

```javascript
// find_all_paths.js
function getInterpPx(pi, loc, step) {
    var locx = loc[0] + Math.max(step[0], 0);
    var locy = loc[1] + Math.max(step[1], 0);
    var zxy = pi.z[locy][locx];

    // 在线性空间插值，然后转换为像素坐标
    if(step[1]) {  // 垂直方向插值
        var dx = (pi.level - zxy) / (pi.z[locy][locx + 1] - zxy);

        // 关键：在对数轴上正确处理插值
        var dxl =
            (dx !== 1 ? (1 - dx) * xa.c2l(pi.x[locx]) : 0) +
            (dx !== 0 ? dx * xa.c2l(pi.x[locx + 1]) : 0);

        return [xa.c2p(xa.l2c(dxl), true),
                ya.c2p(pi.y[locy], true),
                locx + dx, locy];
    } else {  // 水平方向插值
        var dy = (pi.level - zxy) / (pi.z[locy + 1][locx] - zxy);
        var dyl =
            (dy !== 1 ? (1 - dy) * ya.c2l(pi.y[locy]) : 0) +
            (dy !== 0 ? dy * ya.c2l(pi.y[locy + 1]) : 0);

        return [xa.c2p(pi.x[locx], true),
                ya.c2p(ya.l2c(dyl), true),
                locx, locy + dy];
    }
}
```

### 3.3 路径追踪与连接

#### 3.3.1 路径追踪核心逻辑

```javascript
// find_all_paths.js
function makePath(pi, loc, edgeflag, xtol, ytol) {
    var locStr = loc.join(',');
    var mi = pi.crossings[locStr];
    var marchStep = getStartStep(mi, edgeflag, loc);

    // 从起点后退半步找到第一个交点
    var pts = [getInterpPx(pi, loc, [-marchStep[0], -marchStep[1]])];

    // 沿路径追踪
    for(cnt = 0; cnt < 10000; cnt++) {
        // 处理鞍点情况
        if(mi > 20) {
            mi = constants.CHOOSESADDLE[mi][(marchStep[0] || marchStep[1]) < 0 ? 0 : 1];
            pi.crossings[locStr] = constants.SADDLEREMAINDER[mi];
        } else {
            delete pi.crossings[locStr];  // 标记已访问
        }

        // 获取下一步方向
        marchStep = constants.NEWDELTA[mi];

        // 前进半步找到交点，然后完整一步
        pts.push(getInterpPx(pi, loc, marchStep));
        loc[0] += marchStep[0];
        loc[1] += marchStep[1];

        // 检查是否完成循环或到达边界
        var closedLoop = loc[0] === startLoc[0] && loc[1] === startLoc[1];
        if((closedLoop) || (edgeflag && atEdge)) break;

        mi = pi.crossings[loc.join(',')];
    }

    // 根据路径类型分类存储
    if(closedpath) {
        pi.paths.push(pts);  // 闭合路径
    } else {
        pi.edgepaths.push(pts);  // 开放路径（边界路径）
    }
}
```

```javascript
                if(j === i) {
                    // 路径现在闭合了

                } else {
                    pi.edgepaths[j] = edgepathj.concat(pts, edgepathi);
                }

            }
        }

            pi.edgepaths[i] = pts.concat(edgepathi);
        }
    }
}
```

### 3.4 Fill 模式的关键：边界闭合

#### 3.4.1 判断是否需要添加边界

```javascript
// close_boundaries.js
module.exports = function(pathinfo, contours) {
    var pi0 = pathinfo[0];
    var z = pi0.z;

    switch(contours.type) {
        case 'levels':
            // 使用左上角的最小值作为参考
            var edgeVal2 = Math.min(z[0][0], z[0][1]);

            for(i = 0; i < pathinfo.length; i++) {
                var pi = pathinfo[i];
                // 如果没有边界路径，且边界值大于层级，则添加整个边界
                pi.prefixBoundary = !pi.edgepaths.length &&
                    (edgeVal2 > pi.level || pi.starts.length && edgeVal2 === pi.level);
            }
            break;

        case 'constraint':
            // 约束模式的特殊处理
            var boundaryMax = -Infinity;
            var boundaryMin = Infinity;

            // 计算边界的最大/最小值
            for(i = 0; i < nb; i++) {
                boundaryMin = Math.min(boundaryMin, z[i][0], z[i][na - 1]);
                boundaryMax = Math.max(boundaryMax, z[i][0], z[i][na - 1]);
            }

            // 根据操作类型决定是否添加边界
            switch(contours._operation) {
                case '>':
                    if(contoursValue > boundaryMax) {
                        pi0.prefixBoundary = true;
                    }
                    break;
                case '<':
                    if(contoursValue < boundaryMin) {
                        pi0.prefixBoundary = true;
                    }
                    break;
            }
            break;
    }
};
```

#### 3.4.2 Fill 路径生成

```javascript
// plot.js: makeFills 函数
function makeFills(plotgroup, pathinfo, perimeter, contours) {
    var hasFills = contours.coloring === 'fill' ||
                   (contours.type === 'constraint' && contours._operation !== '=');
    var boundaryPath = 'M' + perimeter.join('L') + 'Z';

    // 闭合边界（prefixBoundary 标记在 closeBoundaries 中设置）
    if(hasFills) {
        closeBoundaries(pathinfo, contours);
    }

    var fillitems = fillgroup.selectAll('path').data(hasFills ? pathinfo : []);
    fillitems.each(function(pi) {
        // 构建完整路径：
        // 1. 如果需要，先添加整个边界
        // 2. 沿边界顺时针连接所有开放路径
        // 3. 最后添加所有内部闭合路径
        var fullpath = (pi.prefixBoundary ? boundaryPath : '') +
            joinAllPaths(pi, perimeter);

        d3.select(this).attr('d', fullpath).style('stroke', 'none');
    });
}
```

#### 3.4.3 路径连接算法（关键）

```javascript
// plot.js: joinAllPaths 函数
function joinAllPaths(pi, perimeter) {
    var fullpath = '';
    var startsleft = pi.edgepaths.map(function(v, i) { return i; });
    var newloop = true;

    // 位置判断函数
    function istop(pt) { return Math.abs(pt[1] - perimeter[0][1]) < 0.01; }
    function isbottom(pt) { return Math.abs(pt[1] - perimeter[2][1]) < 0.01; }
    function isleft(pt) { return Math.abs(pt[0] - perimeter[0][0]) < 0.01; }
    function isright(pt) { return Math.abs(pt[0] - perimeter[2][0]) < 0.01; }

    while(startsleft.length) {
        // 添加当前路径（应用平滑）
        addpath = Drawing.smoothopen(pi.edgepaths[i], pi.smoothing);
        fullpath += newloop ? addpath : addpath.replace(/^M/, 'L');
        startsleft.splice(startsleft.indexOf(i), 1);
        endpt = pi.edgepaths[i][pi.edgepaths[i].length - 1];
        nexti = -1;

        // 沿边界移动，寻找下一个路径的起点
        for(cnt = 0; cnt < 4; cnt++) {
            // 确定下一个角点
            if(istop(endpt) && !isright(endpt)) newendpt = perimeter[1];
            else if(isleft(endpt)) newendpt = perimeter[0];
            else if(isbottom(endpt)) newendpt = perimeter[3];
            else if(isright(endpt)) newendpt = perimeter[2];

            // 在当前边上寻找是否有其他路径的起点
            for(possiblei = 0; possiblei < pi.edgepaths.length; possiblei++) {
                var ptNew = pi.edgepaths[possiblei][0];
                // 检查 ptNew 是否在从 endpt 到 newendpt 的线段上
                if(Math.abs(endpt[0] - newendpt[0]) < 0.01) {
                    // 垂直边
                    if(Math.abs(endpt[0] - ptNew[0]) < 0.01 &&
                        (ptNew[1] - endpt[1]) * (newendpt[1] - ptNew[1]) >= 0) {
                        newendpt = ptNew;
                        nexti = possiblei;
                    }
                } else if(Math.abs(endpt[1] - newendpt[1]) < 0.01) {
                    // 水平边
                    if(Math.abs(endpt[1] - ptNew[1]) < 0.01 &&
                        (ptNew[0] - endpt[0]) * (newendpt[0] - ptNew[0]) >= 0) {
                        newendpt = ptNew;
                        nexti = possiblei;
                    }
                }
            }

            endpt = newendpt;
            if(nexti >= 0) break;
            fullpath += 'L' + newendpt;
        }

        i = nexti;
        newloop = (startsleft.indexOf(i) === -1);
        if(newloop) {
            i = startsleft[0];
            fullpath += 'Z';  // 闭合当前循环
        }
    }

    // 添加所有内部闭合路径
    for(i = 0; i < pi.paths.length; i++) {
        fullpath += Drawing.smoothclosed(pi.paths[i], pi.smoothing);
    }

    return fullpath;
}
```

**算法核心思想**：
1. 将所有开放路径（edgepaths）通过边界连接起来
2. 沿边界顺时针移动，寻找下一个路径的起点
3. 使用奇偶填充规则（even-odd rule）确保正确填充
4. 最后添加所有内部闭合路径

### 3.5 曲线平滑（Catmull-Rom 样条）

```javascript
// src/components/drawing/index.js
drawing.smoothopen = function(pts, smoothness) {
    if(pts.length < 3) {
        return 'M' + pts.join('L');
    }
    var path = 'M' + pts[0];
    var tangents = [];

    // 计算每个点的切线向量
    for(i = 1; i < pts.length - 1; i++) {
        tangents.push(makeTangent(pts[i - 1], pts[i], pts[i + 1], smoothness));
    }

    // 使用二次贝塞尔曲线开始
    path += 'Q' + tangents[0][0] + ' ' + pts[1];

    // 使用三次贝塞尔曲线连接中间点
    for(i = 2; i < pts.length - 1; i++) {
        path += 'C' + tangents[i - 2][1] + ' ' + tangents[i - 1][0] + ' ' + pts[i];
    }

    // 使用二次贝塞尔曲线结束
    path += 'Q' + tangents[pts.length - 3][1] + ' ' + pts[pts.length - 1];
    return path;
};

drawing.smoothclosed = function(pts, smoothness) {
    if(pts.length < 3) {
        return 'M' + pts.join('L') + 'Z';
    }
    var path = 'M' + pts[0];
    var pLast = pts.length - 1;

    // 计算所有点的切线（包括首尾相接）
    var tangents = [makeTangent(pts[pLast], pts[0], pts[1], smoothness)];
    for(i = 1; i < pLast; i++) {
        tangents.push(makeTangent(pts[i - 1], pts[i], pts[i + 1], smoothness));
    }
    tangents.push(makeTangent(pts[pLast - 1], pts[pLast], pts[0], smoothness));

    // 使用三次贝塞尔曲线连接所有点
    for(i = 1; i <= pLast; i++) {
        path += 'C' + tangents[i - 1][1] + ' ' + tangents[i][0] + ' ' + pts[i];
    }
    path += 'C' + tangents[pLast][1] + ' ' + tangents[0][0] + ' ' + pts[0] + 'Z';
    return path;
};

// Catmull-Rom 样条切线计算
function makeTangent(prevpt, thispt, nextpt, smoothness) {
    var d1x = prevpt[0] - thispt[0];
    var d1y = prevpt[1] - thispt[1];
    var d2x = nextpt[0] - thispt[0];
    var d2y = nextpt[1] - thispt[1];

    // Catmull-Rom 参数
    var d1a = Math.pow(d1x * d1x + d1y * d1y, CatmullRomExp / 2);
    var d2a = Math.pow(d2x * d2x + d2y * d2y, CatmullRomExp / 2);

    var numx = (d2a * d2a * d1x - d1a * d1a * d2x) * smoothness;
    var numy = (d2a * d2a * d1y - d1a * d1a * d2y) * smoothness;
    var denom1 = 3 * d2a * (d1a + d2a);
    var denom2 = 3 * d1a * (d1a + d2a);

    return [
        [thispt[0] + (denom1 && numx / denom1),
         thispt[1] + (denom1 && numy / denom1)],
        [thispt[0] - (denom2 && numx / denom2),
         thispt[1] - (denom2 && numy / denom2)]
    ];
}
```

**平滑参数说明**：
- `smoothness = 0`: 无平滑，直接连线
- `smoothness = 1`: 标准 Catmull-Rom 样条
- `smoothness > 1`: 更夸张的曲线
- `smoothness < 1`: 更保守的曲线

## 五、关键算法总结

### 5.1 Marching Squares 的优势

1. **高效性**：只检查有等值线穿过的单元格
2. **拓扑正确性**：正确处理鞍点和连通性
3. **可扩展性**：易于扩展到 3D（Marching Cubes）

### 5.2 鞍点处理的关键

Plotly.js 使用**平均值消歧法**：
```javascript
var avg = (corners[0][0] + corners[0][1] + corners[1][0] + corners[1][1]) / 4;
if(val > avg) return (mi === 5) ? 713 : 1114;  // 双峰情况

```

### 5.3 Fill 模式的核心思想

**奇偶填充规则**（Even-Odd Rule）：
1. 从边界开始，沿顺时针方向连接所有开放路径
2. 如果某个区域没有等值线穿过，根据其相对于边界的位置决定是否填充
3. 通过 `prefixBoundary` 确保最外层区域正确填充

**关键代码**：
```javascript
// plot.js: makeBackground
```

## 六、性能优化要点

1. **批量处理**：一次性计算所有层级的交点，而不是逐层计算
2. **惰性删除**：在追踪过程中动态删除已访问的交点
3. **路径缓存**：避免重复计算相同的插值点
4. **平滑优化**：根据平滑系数自适应简化路径

## 七、注意事项

1. **对数轴处理**：插值在线性空间进行，然后转换到对数空间
2. **缺失值处理**：通过 `connectgaps` 参数控制是否填充缺失区域
3. **边界情况**：2×2 网格的特殊处理
4. **阈值范围**：自定义阈值可以在数据范围之外，系统会正确处理
